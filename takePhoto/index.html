const canvas = new fabric.Canvas('c'); let originalWidth = 0; // 用於紀錄原始照片寬度 // 1. 上傳底圖並設定解析度參考 document.getElementById('upload-photo').addEventListener('change', function(e) { const file = e.target.files[0]; const reader = new FileReader(); reader.onload = function(f) { const imgObj = new Image(); imgObj.onload = function() { // 儲存原始尺寸 originalWidth = imgObj.width; const originalHeight = imgObj.height; // 顯示用的畫布大小 (維持手機流暢度) const displayWidth = Math.min(window.innerWidth - 30, 500); const scaleRatio = displayWidth / originalWidth; const displayHeight = originalHeight * scaleRatio; canvas.setWidth(displayWidth); canvas.setHeight(displayHeight); // 將圖片設為底圖，並填滿畫布 fabric.Image.fromURL(f.target.result, function(img) { canvas.setBackgroundImage(img, canvas.renderAll.bind(canvas), { scaleX: canvas.width / img.width, scaleY: canvas.height / img.height }); }); }; imgObj.src = f.target.result; }; reader.readAsDataURL(file); }); // 2. 加入貼圖 (不變) function addSticker(url) { fabric.Image.fromURL(url, function(img) { img.scaleToWidth(canvas.width * 0.3); // 初始大小設為畫布的 30% img.set({ left: 50, top: 50, cornerColor: 'rgba(0,0,255,0.5)', transparentCorners: false, cornerSize: 24 }); canvas.add(img); canvas.setActiveObject(img); }, { crossOrigin: 'anonymous' }); } // 3. 高解析度下載與分享功能 function getHighResDataURL() { // 計算倍率：原始寬度 / 目前畫布寬度 const zoom = originalWidth / canvas.width; return canvas.toDataURL({ format: 'png', multiplier: zoom, // 關鍵：這會讓輸出後的圖片回到原始解析度 quality: 1 }); } document.getElementById('download-btn').onclick = () => { if (!originalWidth) return alert("請先上傳照片"); const dataURL = getHighResDataURL(); const link = document.createElement('a'); link.href = dataURL; link.download = 'high-res-photo.png'; link.click(); }; document.getElementById('share-btn').onclick = async () => { if (!originalWidth) return alert("請先上傳照片"); const dataURL = getHighResDataURL(); const blob = await (await fetch(dataURL)).blob(); const file = new File([blob], 'high-res-photo.png', { type: 'image/png' }); if (navigator.share) { try { await navigator.share({ files: [file], title: '我的高畫質作品', text: '這張是用原始解析度合成的喔！' }); } catch (err) { console.log('分享失敗', err); } } else { alert('瀏覽器不支援分享，請使用儲存功能。'); } };
